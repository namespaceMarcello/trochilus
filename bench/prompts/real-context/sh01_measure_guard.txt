# file: tools/cleanup.lib
# cleanup.lib -- every script kills what it started (docs/LESSONS.md #84).
#
# On 2026-09-17 a busy-machine test left four `yes` processes behind; they ran at 100% for 37
# hours under every native measurement of two days. The cause was not a missing guard on the
# measurements: it was a test that did not clean up after itself. Every script in tools/ sources
# this file (tools/lint.py asks for it) and ends its life through it:
#
#   . tools/cleanup.lib
#   trap cleanup_children EXIT        # whatever this script started dies with it
#   trap 'exit 130' INT TERM          # a signal ends the script (and so runs the line above)
#
#   cleanup_run <command...>          runs a long command in the background and waits for it.
#                                     A shell that waits for a foreground child does not run its
#                                     traps until the child ends: a measuring script told to stop
#                                     went on for the twenty minutes of its ab_modes session
#                                     (docs/LESSONS.md #82). `wait` is interrupted at once.
#   cleanup_children                  kills every descendant of this script, deepest included:
#                                     the engine under `sh -c` under ab_modes.sh, the keep-awake
#                                     helper, a load generator.
#
# Killing the shell that launched a script does not kill the script's children (#82): they are
# found from the process table, by parent, not remembered one by one. On Windows a native
# program started by another native program (the engine under Python) is not in the MSYS
# process table: taskkill takes the whole tree below each descendant.
# tools/test_cleanup.sh proves it, and proves that without the trap the child stays.

# every descendant of pid $1, from `ps -ef` (PID in column 2, PPID in column 3, on MSYS and Linux)
cleanup_descendants() {
  ps -ef 2> /dev/null | awk -v root="$1" '
    { pid[NR] = $2; ppid[NR] = $3 }
    END {
      want[root] = 1
      for (changed = 1; changed; ) {
        changed = 0
        for (i = 1; i <= NR; i++) if ((ppid[i] in want) && !(pid[i] in want)) { want[pid[i]] = 1; changed = 1 }
      }
      for (i = 1; i <= NR; i++) if ((pid[i] in want) && pid[i] != root) print pid[i]
    }'
}

cleanup_children() {
  CLEANUP_LIST=$(cleanup_descendants $$)
  if command -v taskkill > /dev/null 2>&1; then
    # MSYS: the Windows pid of each descendant (ps -l: PID PPID PGID WINPID ...), and its tree
    for CLEANUP_PID in $CLEANUP_LIST; do
      CLEANUP_WIN=$(ps -l 2> /dev/null | awk -v p="$CLEANUP_PID" '{ for (i = 1; i <= 2; i++) if ($i == p) { print $(i + 3); exit } }')
      [ -z "$CLEANUP_WIN" ] || taskkill //F //T //PID "$CLEANUP_WIN" > /dev/null 2>&1 || true
    done
  fi
  for CLEANUP_PID in $CLEANUP_LIST; do kill "$CLEANUP_PID" 2> /dev/null || true; done
  for CLEANUP_PID in $CLEANUP_LIST; do
    kill -0 "$CLEANUP_PID" 2> /dev/null || continue
    kill -9 "$CLEANUP_PID" 2> /dev/null || true
  done
}

cleanup_run() {
  "$@" &
  CLEANUP_RUN=$!
  wait $CLEANUP_RUN
}
# file: tools/measure_guard.lib
# measure_guard.lib -- functions sourced by the scripts that measure on a still machine
# (prefill_context.sh, decode_context.sh, threads_phase.sh, remeasure.sh). What a long
# measurement needs and nobody gave it (docs/LESSONS.md #82, #84):
#
#   measure_begin <name>   one measurement at a time: a second script finds the lock of the first
#                          and refuses to start (exit 5), instead of running its models beside
#                          it; the lock of a script that died is taken over. Then the caller's
#                          measure_ready, when it defines one (the wait for a binary Smart App
#                          Control holds: up to an hour, and nobody else waits for it), then the
#                          machine's marker is taken (below), nothing this project started is still
#                          running (tools/orphans.sh: a forgotten load generator, an engine,
#                          another script) or it refuses (exit 6), and the machine is kept awake
#                          for as long as the measurement lasts (tools/stay_awake.ps1, Windows).
#                          The caller sets its traps (below) before calling it.
#   measure_machine <name> after measure_begin: names the containers left running, gives the
#                          Docker VM's file cache back to Windows, and sets AB_GUARD.
#   measure_still <name>   before the first run: the CPU is not busy with somebody else's work,
#                          or it waits, then refuses (exit 3).
#   measure_end            ends what the script started (tools/cleanup.lib), releases the lock
#                          and the marker; the caller puts it in its EXIT trap, and `exit 130` in
#                          its INT and TERM traps.
#
# The machine is shared with other Claude windows, each on its own project (docs/LESSONS.md #73,
# #87). Their rule, since 2026-09-23: whoever measures times writes one line ("project: what") to
# ~/.claude/macchina-ferma for as long as it measures, and the others' correctness work (tests,
# mutations) pauses while the file is there; a file older than 6 hours is stale. So the containers
# of the other windows are not stopped and not waited for: they stay up, idle, and a window whose
# work does not pause shows in the CPU check before every run. The marker is written with
# noclobber (one writer wins), refreshed before every run, and removed only by its writer.
#
# A measurement is stopped by killing the pid in build/.measuring.lock/pid: the script runs its
# long steps through cleanup_run, so the signal is served at once, and its EXIT trap takes the
# ab_modes session and the engine run under it along.
. tools/cleanup.lib
MEASURE_LOCK=build/.measuring.lock
MACHINE_MARKER=${MACHINE_MARKER:-$HOME/.claude/macchina-ferma}
MACHINE_MARKER_MAX_MIN=360
MACHINE_MARKER_POLL=${MACHINE_MARKER_POLL:-30}
MACHINE_MARKER_WAIT_MAX=${MACHINE_MARKER_WAIT_MAX:-21600}

# the marker is there, and younger than 6 hours
marker_fresh() {
  [ -f "$MACHINE_MARKER" ] && [ -n "$(find "$MACHINE_MARKER" -mmin -$MACHINE_MARKER_MAX_MIN 2> /dev/null)" ]
}

# a marker of this project whose script is gone: the pid in its line no longer runs (one of
# another checkout's live measurements is waited for like any other window's)
marker_leftover() {
  MARK_PID=$(head -1 "$MACHINE_MARKER" 2> /dev/null | sed -n 's/^trochilus: .* (pid \([0-9][0-9]*\), since .*/\1/p')
  [ -n "$MARK_PID" ] && ! kill -0 "$MARK_PID" 2> /dev/null
}

# measure_mark <name>: take the marker, waiting while another window holds it, for 6 hours at
# most (then exit 5, as for our own lock). A stale marker and a leftover of ours are removed
# first; the line says when it was taken.
measure_mark() {
  mkdir -p "$(dirname "$MACHINE_MARKER")"
  T0=$(date +%s)
  NEXT=0
  while :; do
    if [ -f "$MACHINE_MARKER" ] && { ! marker_fresh || marker_leftover; }; then
      rm -f "$MACHINE_MARKER"
    fi
    MEASURE_MARK_LINE="trochilus: $1 (pid $$, since $(date '+%Y-%m-%d %H:%M'))"
    # set before the create: a TERM served right after it still gives the marker back, and
    # measure_end removes it only while its line is this one (it holds our pid)
    MEASURE_MARK_MINE=1
    if (set -C; echo "$MEASURE_MARK_LINE" > "$MACHINE_MARKER") 2> /dev/null; then
      echo "$1: $MACHINE_MARKER taken, the other windows pause their work until the end"
      return 0
    fi
    WAITED=$(( $(date +%s) - T0 ))
    if [ $WAITED -ge $MACHINE_MARKER_WAIT_MAX ]; then
      echo "$1: another window still measures times after $((WAITED / 60)) min ($(head -1 "$MACHINE_MARKER" 2> /dev/null)): not starting"
      exit 5
    fi
    if [ $WAITED -ge $NEXT ]; then
      echo "$1: another window measures times ($(head -1 "$MACHINE_MARKER" 2> /dev/null)), waiting since $((WAITED / 60)) min"
      NEXT=$((NEXT + 300))
    fi
    cleanup_run sleep $MACHINE_MARKER_POLL
  done
}
